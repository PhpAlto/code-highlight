# Edge cases for DotEnv coverage

# Empty lines


# Export keyword
export DATABASE_URL=postgres://localhost/db

# Keys with underscores and numbers
MY_VAR_123=value
_PRIVATE=secret
VAR_WITH_MANY_UNDERSCORES_HERE=test

# Quoted strings - single quotes
SINGLE_QUOTED='hello world'
SINGLE_WITH_ESCAPE='can'
EMPTY_SINGLE=''

# Quoted strings - double quotes
DOUBLE_QUOTED="hello world"
DOUBLE_WITH_NEWLINE="line1\nline2"
DOUBLE_WITH_TAB="before\tafter"
DOUBLE_WITH_ESCAPE="quote: \" backslash: \\"
EMPTY_DOUBLE=""

# Variable references with braces
REF_WITH_BRACES=${HOME}/app
NESTED_REF=${BASE_PATH}/${SUB_DIR}
MULTIPLE_REFS=${HOST}:${PORT}

# Variable references without braces
REF_WITHOUT_BRACES=$HOME/app
SIMPLE_REF=$USER

# Boolean values
BOOL_TRUE=true
BOOL_FALSE=false
BOOL_TRUE_CAPS=TRUE
BOOL_FALSE_CAPS=FALSE

# Null value
NULL_VALUE=null
NULL_CAPS=NULL

# Numbers
INT_VALUE=42
INT_NEGATIVE=-123
FLOAT_VALUE=3.14
FLOAT_SCIENTIFIC=1.5e10
FLOAT_WITH_DECIMALS=0.123456

# Mixed content in values
MIXED_TEXT_AND_VAR=prefix_${VAR}_suffix
MIXED_MULTIPLE=text1_$VAR1_text2_${VAR2}_text3
URL_WITH_VAR=https://$HOST:8080/path

# Plain unquoted strings
PLAIN_TEXT=some_plain_text
PATH=/usr/local/bin
URL=https://example.com

# Whitespace variations
SPACES_AROUND=value_with_spaces
TABS_AROUND=value_with_tabs
NO_SPACES=no_spaces

# Key without value (incomplete)
INCOMPLETE_KEY

# Comment after export
export COMMENTED=value # inline comment not supported
