; Edge cases for INI coverage
# Both comment styles

; Empty section
[]

[Section1]
; Basic key-value pairs
key1=value1
key2 = value2
key_with_underscores=test
KeyWithCaps=Value

[Section.Subsection]
; Quoted strings - double quotes
double_quoted="hello world"
double_with_escape="quote: \" backslash: \\"
double_with_newline="line1\nline2"
empty_double=""

[Section.Subsection.Deep]
; Quoted strings - single quotes
single_quoted='hello world'
single_with_escape='can'\''t'
empty_single=''

[Environment Variables]
; ${VAR} format
var_with_braces=${HOME}/path
var_multiple=${HOST}:${PORT}

; %VAR% format
var_with_percent=%USERPROFILE%\path
var_percent_multiple=%HOST%:%PORT%

[Boolean Values]
; Various boolean formats
bool_true=true
bool_false=false
bool_yes=yes
bool_no=no
bool_on=on
bool_off=off
BOOL_TRUE_CAPS=TRUE
BOOL_FALSE_CAPS=FALSE

[Null Values]
null_lowercase=null
null_uppercase=NULL
null_none=none
null_none_caps=NONE

[Numbers]
; Integers
int_positive=42
int_negative=-123
int_zero=0

; Floats
float_simple=3.14
float_negative=-2.71
float_scientific=1.5e10
float_scientific_neg=2.3e-5

[Mixed Content]
; URLs and paths
url=https://example.com:8080/path
path=/usr/local/bin
windows_path=C:\Program Files\App

; Unquoted strings with special chars
unquoted_with_space=value with spaces
unquoted_with_underscore=value_with_underscore
unquoted_with_dash=value-with-dash

[Whitespace Handling]
; Leading/trailing spaces
spaces_around = value_with_spaces
tabs_around	=	value_with_tabs
no_spaces=no_spaces

; Empty values
empty_value=
empty_value_with_spaces =

[Inline Comments]
; Note: Inline comments after values are not standard INI
key_with_text=value ; this is not a comment, it's part of the value

[Special Characters in Section Names]
[Section-with-dash]
[Section_with_underscore]
[Section.with.dots]
[Section with spaces]

[Escaped Characters]
value_with_backslash=C:\\path\\to\\file
value_with_quote=value\"with\"quotes

[Array-like Values]
array_style[]=value1
array_style[]=value2
indexed_array[0]=first
indexed_array[1]=second
